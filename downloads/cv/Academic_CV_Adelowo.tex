% ── Projects ─────────────────────────────────────────────────────────────────
\cvsec{Selected Research \& Computational Projects}

\noindent\textbf{\href{https://github.com/AdebanjiAdelowo/fem-cylinder-flow}{Finite-Element Solver for 2D Incompressible Flow: Sch\"afer--Turek Cylinder Benchmarks}}
\begin{cvitems}
  \item Implemented Taylor--Hood finite-element methods for two-dimensional incompressible Navier--Stokes flow past a cylinder, with manufactured-solution convergence checks.
  \item Compared the steady 2D-1 and periodic 2D-2 regimes with published benchmark quantities; including drag, lift and flow dynamics.
\end{cvitems}

\noindent\textbf{\href{https://github.com/AdebanjiAdelowo/Master-Thesis}{Optimal Mixing of Passive Scalars: M.Sc.\ Thesis and Numerical Convergence Study}}
\begin{cvitems}
  \item Studied energy- and enstrophy-constrained optimal stirring and implemented the Lin--Thiffeault--Doering greedy strategy with a pseudo-spectral solver.
  \item Extended the numerical resolution study to $N=512$; identified changes in fitting windows as a major explanation of apparent resolution dependence. 
\end{cvitems}

\noindent\textbf{\href{https://github.com/AdebanjiAdelowo/photoacoustic-reconstruction}{Photoacoustic Reconstruction: Model-Based vs.\ Learned Inversion}}
\begin{cvitems}
  \item Built a synthetic sparse-view acoustic inversion pipeline comparing time reversal, Tikhonov regularisation and learned U-Net refinement.
  \item Across 400 held-out images and five network seeds, learned refinement improved PSNR over calibrated time reversal by 4.9\,dB (16 sensors) and 3.2\,dB (64 sensors).
\end{cvitems}

\noindent\textbf{\href{https://github.com/AdebanjiAdelowo/lorenz96-data-assimilation}{Ensemble Kalman Filtering for the Chaotic Lorenz-96 System}}
\begin{cvitems}
  \item Implemented ensemble Kalman filtering with cyclic localisation and inflation for partially observed Lorenz-96 dynamics.
  \item Cross-checked the assimilation implementation against a linear-Gaussian Kalman-filter reference and evaluated localisation, inflation and ensemble-size effects.
\end{cvitems}

\noindent\textbf{\href{https://github.com/AdebanjiAdelowo/neural-surrogate-burgers}{ROM, Hyper-Reduction and Neural Operators for Viscous Burgers' Equation}}
\begin{cvitems}
  \item Compared a pseudo-spectral reference solver with POD--Galerkin reduced-order modelling, DEIM hyper-reduction, an MLP and a Fourier Neural Operator.
  \item The POD-ROM reached 0.152\% relative $L^2$ error on the in-distribution benchmark, while the FNO reached 0.90\%.
  \item Distinguished basis projection error, reduced-dynamics stability, hyper-reduction efficiency and learned-operator generalisation.
\end{cvitems}
